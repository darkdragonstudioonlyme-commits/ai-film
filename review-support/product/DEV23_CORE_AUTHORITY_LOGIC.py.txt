def _path_parts(path,prefix):

def _get_path(value,parts):
    cur=value
    for part in parts:
        require(type(cur) is dict and part in cur,10,'TEMPLATE_PATH');cur=cur[part]

def validate_lineage(store,ref,suite_ref,case_id,index,slot_ref):
    hash_value(ref);v=store.get('lab_derived_stage_authority',ref)
    req={'role','schema_version','withdrawn','source_kind','base_manifest_ref','slot_ref','suite_ref','execution_id','case_id','procedure_digest','stage_index','route','producer_handoff_refs','producer_evidence_digests','materialized_values_digest','plan_template_ref','native_binding_template_ref','native_binding_ref','execution_plan_ref','approval_ref','checkpoint_payload_ref','restore_envelope_ref','immutable_partition_digest','parent_policy_digest','parent_generation','new_generation','created_at','expires_at'}
    require(set(v)==req and v['schema_version']==1 and v['withdrawn'] is False and v['source_kind']=='LAB',15,'DERIVED_LINEAGE_SCHEMA')
    require(v['suite_ref']==suite_ref and v['case_id']==case_id and v['stage_index']==index and v['slot_ref']==slot_ref and v['base_manifest_ref']==v['immutable_partition_digest'],16,'DERIVED_LINEAGE_SCOPE')
    for key in ('base_manifest_ref','slot_ref','materialized_values_digest','plan_template_ref','execution_plan_ref','approval_ref','immutable_partition_digest','parent_policy_digest'):hash_value(v[key])
    for key in ('native_binding_template_ref','native_binding_ref','checkpoint_payload_ref','restore_envelope_ref'):
        if v[key] is not None:hash_value(v[key])
    require(type(v['parent_generation']) is int and v['new_generation']==v['parent_generation']+1 and v['new_generation']<=store.generation,16,'DERIVED_LINEAGE_GENERATION')
    plan_doc=store.get('execution_plan',v['execution_plan_ref']);plan=plan_doc.get('plan');s=check_plan(plan);require(plan.get('approval_ref')==v['approval_ref'] and s['purpose']==v['route'],16,'DERIVED_LINEAGE_PLAN')
    store.get('approval',v['approval_ref'])
    if v['native_binding_ref'] is not None:store.get('native_binding',v['native_binding_ref'])
    if v['checkpoint_payload_ref'] is not None:store.get('checkpoint_payload',v['checkpoint_payload_ref'])
    return v

def resolve_current_lineage(store,suite_ref,case_id,index,slot_ref):
    matches=[]
    for ref in sorted(_pins(store,'lab_derived_stage_authority')):
        item=store.get('lab_derived_stage_authority',ref)
        if item.get('suite_ref')==suite_ref and item.get('case_id')==case_id and item.get('stage_index')==index and item.get('slot_ref')==slot_ref:
            validate_lineage(store,ref,suite_ref,case_id,index,slot_ref);matches.append((ref,item))
    require(len(matches)<=1,15,'DERIVED_LINEAGE_AMBIGUOUS');return matches[0] if matches else None

def lineage_revocation_refs(store,lineage_ref):
    v=store.get('lab_derived_stage_authority',lineage_ref);refs={lineage_ref,v['execution_plan_ref'],v['approval_ref']}
    for k in ('native_binding_ref','checkpoint_payload_ref'):

def prepare_lineage_revocation(parent_policy_bytes,store,lineage_ref,event_identity):
    lineage=store.get('lab_derived_stage_authority',lineage_ref);manifest=store.get('lab_base_manifest',lineage['base_manifest_ref'])
    immutable={row['ref'] for row in manifest['entries']};withdrawals=sorted(lineage_revocation_refs(store,lineage_ref))

    gref,g=generation_record(parent_policy_bytes,lineage['immutable_partition_digest'],{},event_identity,withdrawals)

def validate_stage_authority_record(store,suite_ref,suite,case_id,index,proc,request,stage):
    checked=validate_request_row(store,suite_ref,suite,case_id,proc,index,request);mode=checked['authority_mode']
    require(stage.get('authority_mode')==mode and stage.get('base_manifest_ref')==checked['base_manifest_ref'],16,'LAB_STAGE_AUTHORITY_MODE')
    require(stage.get('authority_ref')==checked.get('authority_ref',checked.get('derivation_slot_ref',checked.get('reconciliation_slot_ref'))),16,'LAB_STAGE_AUTHORITY_REF')
    if mode=='CONCRETE_PRE_V03':
        require(stage.get('authority_lineage_ref') is None,16,'LAB_STAGE_LINEAGE')
        plan=store.get('execution_plan',checked['plan_ref']).get('plan');require(check_plan(plan)['purpose']==checked['route'] and plan['plan_digest']==stage['plan_digest'],16,'LAB_STAGE_AUTHORITY_PLAN')
    elif mode=='ENTRY_PROBE_AUTHORITY':
        require(stage.get('authority_lineage_ref') is None,16,'LAB_STAGE_LINEAGE');probe=store.get('entry_probe_set',checked['entry_probe_ref']);plan=store.get('execution_plan',probe['plan_ref']).get('plan')
        require(plan['plan_digest']==stage['plan_digest'],16,'LAB_STAGE_AUTHORITY_PLAN')
    elif mode=='STAGE_DERIVED':
        ref=stage.get('authority_lineage_ref');hash_value(ref);lineage=validate_lineage(store,ref,suite_ref,case_id,index,checked['derivation_slot_ref'])
        require(lineage['execution_plan_ref'] in _pins(store,'execution_plan'),16,'LAB_STAGE_AUTHORITY_PLAN');plan=store.get('execution_plan',lineage['execution_plan_ref']).get('plan');require(plan['plan_digest']==stage['plan_digest'],16,'LAB_STAGE_AUTHORITY_PLAN')
    else:
        require(mode=='FENCE_BOUND_RECONCILIATION' and stage.get('authority_lineage_ref') is None,16,'LAB_STAGE_LINEAGE')
        _,manifest=manifest_for_suite(store,suite_ref);slot=validate_slot(store,checked['reconciliation_slot_ref'],suite,case_id,index,proc,authority_rule(case_id,index,checked['route'],proc.procedure_digest),manifest)
        matches=[r for r in slot['reconciliation_plan_refs'] if store.get('execution_plan',r).get('plan',{}).get('plan_digest')==stage['plan_digest']]
        require(len(matches)==1,16,'LAB_STAGE_AUTHORITY_PLAN')
    return True

def validate_handoff(store,ref,suite_ref,suite,case_id,proc,dep):
    hash_value(ref);h=store.get('lab_stage_state_handoff',ref)
    req={'role','schema_version','withdrawn','source_kind','host_id','owner_sid','execution_id','suite_ref','case_id','procedure_digest','stage_index','route','plan_digest','stage_ref','state','evidence_digest','actual','timestamp_utc'}
    require(set(h)==req and h['schema_version']==1 and h['withdrawn'] is False and h['source_kind']=='LAB',15,'STAGE_HANDOFF_SCHEMA')
    require(h['host_id']==suite['host_id'] and h['owner_sid']==suite['owner_sid'] and h['execution_id']==suite['execution_id'] and h['suite_ref']==suite_ref and h['case_id']==case_id and h['procedure_digest']==proc.procedure_digest,16,'STAGE_HANDOFF_SCOPE')
    require(h['stage_index']==dep['producer_stage_index'] and h['route']==dep['required_route'] and h['state']==dep['required_state'],16,'STAGE_HANDOFF_DEPENDENCY')
    hash_value(h['plan_digest']);hash_value(h['stage_ref']);hash_value(h['evidence_digest']);instant(h['timestamp_utc']);require(type(h['actual']) is dict,15,'STAGE_HANDOFF_ACTUAL')
    for name in dep['provides']: require(name in h['actual'],16,'STAGE_HANDOFF_SELECTOR')

def resolve_reconciliation_plan(store,suite_ref,suite,case_id,index,proc,slot_ref,fence):
    manifest_ref,manifest=manifest_for_suite(store,suite_ref);rule=authority_rule(case_id,index,proc.routes[index],proc.procedure_digest)
    slot=validate_slot(store,slot_ref,suite,case_id,index,proc,rule,manifest);require(rule['authority_mode']=='FENCE_BOUND_RECONCILIATION',10,'RECONCILIATION_SLOT_MODE')
    require(type(fence) is dict and bool(fence),11,'FENCE_BOUND_STATE_REQUIRED');matches=[]
    for ref in slot['reconciliation_plan_refs']:
        doc=store.get('execution_plan',ref);plan=doc.get('plan');s=check_plan(plan)
        if s['purpose']!='RECONCILIATION_ONLY' or s['host_id']!=suite['host_id'] or s['owner_sid']!=suite['owner_sid']:continue
        original_ref=s['refs'].get('original_plan')
        if original_ref is None:continue
        original=store.get('original_plan',original_ref).get('plan');old=check_plan(original)
        if fence.get('plan_digest')==original['plan_digest'] and all(fence.get(k)==old[k] for k in ('run_id','host_id','owner_sid')):matches.append((ref,plan))
    require(len(matches)==1,15,'RECONCILIATION_PLAN_SELECTION');return {'plan_ref':matches[0][0],'plan':matches[0][1],'slot_ref':slot_ref,'base_manifest_ref':manifest_ref}
