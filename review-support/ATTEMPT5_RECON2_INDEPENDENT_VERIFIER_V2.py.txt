#!/usr/bin/env python3
from pathlib import Path
import hashlib, json, os, stat, subprocess, time

ROOT=Path("/home/dragon/ai-film-dev")
TX=ROOT/"run-evidence/prodlike-dev23-corrected-attempt5-reconciliation2-20260924/transaction-receipt.json"

def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()

def snap(p):
    p=Path(p)
    if not p.exists() and not p.is_symlink(): return {"type":"absent"}
    if p.is_symlink(): return {"type":"symlink","target":os.readlink(p)}
    if p.is_file():
        s=p.stat()
        return {"type":"file","sha256":sha(p),"bytes":s.st_size,"mode":format(stat.S_IMODE(s.st_mode),"o")}
    rows={}
    for x in sorted(p.rglob("*")):
        rel=x.relative_to(p).as_posix()
        if x.is_symlink(): rows[rel]={"type":"symlink","target":os.readlink(x)}
        elif x.is_file():
            s=x.stat(); rows[rel]={"type":"file","sha256":sha(x),"bytes":s.st_size,"mode":format(stat.S_IMODE(s.st_mode),"o")}
        elif x.is_dir(): rows[rel]={"type":"dir","mode":format(stat.S_IMODE(x.stat().st_mode),"o")}
    return {"type":"dir","entries":rows}
